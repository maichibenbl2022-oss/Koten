local KEY_TAIL     = ".  "
local KEY_DURATION = 3600
local KEY_FILE     = "HM_key_v12.dat"
local unlocked = false

local function readKeyRaw()
    local ok, d = pcall(function()
        if readfile then
            if isfile then
                if isfile(KEY_FILE) then 
                    return readfile(KEY_FILE) 
                end
            end
        end
        return nil
    end)
    return ok and d or nil
end

local function writeKeyRaw(s) 
    pcall(function() 
        if writefile then 
            writefile(KEY_FILE, s) 
        end 
    end) 
end

local function wipeKey() 
    pcall(function() 
        if writefile then 
            writefile(KEY_FILE, "") 
        end 
    end) 
end

local function validateKey(k)
    if type(k) ~= "string" then 
        return false 
    end
    
    if #k < 4 then 
        return false 
    end
    
    if string.sub(k, -3) ~= KEY_TAIL then 
        return false 
    end
    
    return true
end

local function keyValidNow()
    local raw = readKeyRaw()
    
    if not raw then 
        return false 
    end
    
    if raw == "" then 
        return false 
    end
    
    local a, b = string.find(raw, "|", 1, true)
    
    if not a then 
        return false 
    end
    
    local k  = string.sub(raw, 1, a - 1)
    local ts = tonumber(string.sub(raw, b + 1))
    
    if not k then 
        return false 
    end
    
    if not ts then 
        return false 
    end
    
    if not validateKey(k) then 
        return false 
    end
    
    if (os.time() - ts) >= KEY_DURATION then 
        return false 
    end
    
    return true
end

local function setupM2DynamicListener(p)
    if p == LocalPlayer then 
        return 
    end
    
    local function onChar(char)
        if not char then 
            return 
        end
        
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        
        if not hum then 
            return 
        end
        
        if not root then 
            return 
        end
        
        hum.AnimationPlayed:Connect(function(track)
            if not isBlockActive then 
                return 
            end
            
            local animId = track.Animation and track.Animation.AnimationId
            
            if animId == M2_TargetID then
                local myChar = LocalPlayer.Character
                
                if myChar then
                    if myChar:FindFirstChild("HumanoidRootPart") then
                        local myPos = myChar.HumanoidRootPart.Position
                        local cPos = root.Position
                        local dx = cPos.X - myPos.X
                        local dy = cPos.Y - myPos.Y
                        local dz = cPos.Z - myPos.Z
                        
                        if (dx * dx + dy * dy + dz * dz) <= COMBAT_RANGE_SQ then
                            activeM2Count = activeM2Count + 1
                            rawUnblock()
                            
                            local cleaned = false
                            local stoppedConn, endedConn, diedConn
                            local timeoutTask
                            
                            local function cleanup()
                                if cleaned then 
                                    return 
                                end
                                
                                cleaned = true
                                activeM2Count = math.max(0, activeM2Count - 1)
                                
                                if stoppedConn then 
                                    stoppedConn:Disconnect() 
                                end
                                
                                if endedConn then 
                                    endedConn:Disconnect() 
                                end
                                
                                if diedConn then 
                                    diedConn:Disconnect() 
                                end
                                
                                if timeoutTask then 
                                    task.cancel(timeoutTask) 
                                end
                            end
                            
                            stoppedConn = track.Stopped:Connect(cleanup)
                            endedConn = track.Ended:Connect(cleanup)
                            diedConn = hum.Died:Connect(cleanup)
                            
                            local animDuration = (track.Length > 0 and track.Length) or 1.5
                            timeoutTask = task.delay(animDuration + 0.1, cleanup)
                        end
                    end
                end
            end
        end)
    end
    
    if p.Character then 
        onChar(p.Character) 
    end
    
    p.CharacterAdded:Connect(onChar)
end

    tr.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            drag = true
            local rx = input.Position.X - tr.AbsolutePosition.X
            setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
        elseif input.UserInputType == Enum.UserInputType.Touch then
            drag = true
            local rx = input.Position.X - tr.AbsolutePosition.X
            setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    
    UserInputService.InputChanged:Connect(function(input)
        if drag then
            if input.UserInputType == Enum.UserInputType.MouseMovement then
                local rx = input.Position.X - tr.AbsolutePosition.X
                setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
            elseif input.UserInputType == Enum.UserInputType.Touch then
                local rx = input.Position.X - tr.AbsolutePosition.X
                setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
            end
        end
    end)
    
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            drag = false
        elseif input.UserInputType == Enum.UserInputType.Touch then
            drag = false
        end
    end)
    
    return y + 40
end

local function Valid(p)
    if not p then 
        return false 
    end
    if p == LocalPlayer then 
        return false 
    end
    if Allies[p.Name] then 
        return false 
    end
    local c = p.Character
    if not c then 
        return false 
    end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    if not r then 
        return false 
    end
    if not h then 
        return false 
    end
    if h.Health <= 0 then 
        return false 
    end
    return true
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then 
        return nil 
    end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then 
                    bd = d
                    best = p 
                end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name then 
        return nil 
    end
    if name == "" then 
        return nil 
    end
    name = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            if string.find(string.lower(p.Name), name, 1, true) then 
                return p 
            end
        end
    end
    return nil
end

local function GetTool()
    local c = LocalPlayer.Character
    if not c then 
        return nil 
    end
    local t = c:FindFirstChildOfClass("Tool")
    if t then 
        return t 
    end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, x in ipairs(bp:GetChildren()) do
            if x:IsA("Tool") then 
                x.Parent = c
                return x 
            end
        end
    end
    return nil
end

local function HitAll(target)
    local c = LocalPlayer.Character
    if not c then 
        return 
    end
    for _, t in ipairs(c:GetChildren()) do
        if t:IsA("Tool") then 
            pcall(function() 
                t.Enabled = true
                t:Activate() 
            end) 
        end
    end
    for _, r in ipairs(Remotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer() end)
    end
    for _, r in ipairs(ManualRemotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer("Attack") end)
        pcall(function() r:FireServer() end)
    end
    if clientSideHitbox and target then 
        pcall(function() 
            clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) 
        end) 
    end
    if combatRemote then 
        pcall(function() combatRemote:FireServer("M1") end) 
        pcall(function() combatRemote:FireServer("Attack") end) 
    end
end

task.spawn(function()
    while true do
        task.wait(0.02)
        if State.ORLock and not State.Farm then
            pcall(function()
                if not Valid(Config.Target) then
                    local name = targetBox.Text
                    if name == "" then
                        Config.Target = FindNearest(Config.ORRange)
                    else
                        Config.Target = FindByName(name)
                    end
                    if not Config.Target then
                        return
                    end
                end
                
                local tc = Config.Target.Character
                local tRoot = tc and tc:FindFirstChild("HumanoidRootPart")
                local myC = LocalPlayer.Character
                local myRoot = myC and myC:FindFirstChild("HumanoidRootPart")
                
                if not tRoot then
                    return
                end
                if not myRoot then
                    return
                end
                
                local tp = tRoot.Position
                local now = tick()

                if Config.ORMode == "DASH" then
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        dashPhase = "attack"
                        dashTimer = now + Config.DashAttackDuration
                        dashNextHit = 0
                    elseif dashPhase == "attack" then
                        myRoot.CFrame = CFrame.lookAt(tp - tRoot.CFrame.LookVector * Config.DashOrbitRadius, tp)
                        if now >= dashNextHit then
                            dashNextHit = now + Config.DashHitRate
                            HitAll(Config.Target)
                        end
                        if now >= dashTimer then
                            myRoot.CFrame = dashSavedPos
                            dashPhase = "wait"
                            dashTimer = now + Config.DashWaitDuration
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then
                            dashPhase = "idle"
                        end
                    end
                elseif Config.ORMode == "TPLOOP" then
                    if dashPhase == "idle" then
                        dashSavedPos = myRoot.CFrame
                        myRoot.CFrame = CFrame.lookAt(tp - tRoot.CFrame.LookVector * Config.ORDistance, tp)
                        dashPhase = "attack"
                        dashTimer = now + Config.TPLoopTime
                    elseif dashPhase == "attack" then
                        HitAll(Config.Target)
                        if now >= dashTimer then
                            myRoot.CFrame = dashSavedPos
                            dashPhase = "wait"
                            dashTimer = now + 1
                        end
                    elseif dashPhase == "wait" then
                        if now >= dashTimer then
                            dashPhase = "idle"
                        end
                    end
                else
                    myRoot.CFrame = CFrame.lookAt(tp - tRoot.CFrame.LookVector * Config.ORDistance + Vector3.new(0, Config.ORHeight, 0), tp)
                    HitAll(Config.Target)
                end
            end)
        end
    end
end)

RunService.Heartbeat:Connect(function()
    pcall(function()
        if isBlockActive then
            if activeM2Count > 0 then
                rawUnblock()
            else
                local myChar = LocalPlayer.Character
                if myChar then
                    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
                    if myRoot then
                        local myPos = myRoot.Position
                        local m1Detected = false
                        for _, p in ipairs(Players:GetPlayers()) do
                            if p ~= LocalPlayer then
                                if p.Character then
                                    local cRoot = p.Character:FindFirstChild("HumanoidRootPart")
                                    local cHum = p.Character:FindFirstChild("Humanoid")
                                    if cRoot then
                                        if cHum then
                                            if cHum.Health > 0 then
                                                local dx = cRoot.Position.X - myPos.X
                                                local dy = cRoot.Position.Y - myPos.Y
                                                local dz = cRoot.Position.Z - myPos.Z
                                                if (dx*dx+dy*dy+dz*dz) <= COMBAT_RANGE_SQ then
                                                    local anims = cHum:GetPlayingAnimationTracks()
                                                    for i = 1, #anims do
                                                        if M1_Hash[anims[i].Animation.AnimationId] then
                                                            m1Detected = true
                                                            break
                                                        end
                                                    end
                                                end
                                            end
                                        end
                                    end
                                end
                            end
                            if m1Detected then
                                break
                            end
                        end
                        if m1Detected then
                            lastM1Time = tick()
                            rawBlock()
                        else
                            if isBlocking then
                                if (tick() - lastM1Time) > BUFFER_TIME then
                                    rawUnblock()
                                end
                            end
                        end
                    end
                end
            end
        end
        
        if Config.AutoReset then
            local myHum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if myHum then
                if myHum.Health > 0 then
                    if State.ORLock or State.Spam then
                        if tick() - lastResetTimer >= Config.ResetTime then
                            if resetEvent then
                                pcall(function()
                                    resetEvent:FireServer()
                                end)
                            end
                            lastResetTimer = tick()
                        end
                    end
                end
            end
        end
        
        if State.GodMode then
            local c = LocalPlayer.Character
            if c then
                local h = c:FindFirstChildOfClass("Humanoid")
                if h then
                    h.MaxHealth = math.huge
                    h.Health = math.huge
                end
            end
        end
        
        if State.NoClip then
            local c = LocalPlayer.Character
            if c then
                for _, part in ipairs(c:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end
        end
        
        if State.TP then
            if tpTarget then
                local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local tr = tpTarget.Character and tpTarget.Character:FindFirstChild("HumanoidRootPart")
                if mr then
                    if tr then
                        if tick() - tpStartTime < 0.5 then
                            mr.CFrame = tr.CFrame * CFrame.new(0, 0, Config.TPDistance)
                        else
                            State.TP = false
                        end
                    end
                end
            end
        end
    end)
end)

local dragging, dragStart, startPos

HomeBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = HomeBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    elseif input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = HomeBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

    if dragging then
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            HomeBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)

            local delta = input.Position - dragStart
            HomeBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end
end)
